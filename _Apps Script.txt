// ========== ГЛАВНАЯ ТОЧКА ВХОДА ==========
function doGet(e) {
  const action = e?.parameter?.action;
  
  if (action === 'full') return getFullData();
  if (action === 'params') return getTournamentParams();
  if (action === 'teams') return getTeamsData();
  
  return ContentService.createTextOutput(JSON.stringify({ error: 'Unknown action' }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ========== 1. ПОЛУЧЕНИЕ ПАРАМЕТРОВ АКТИВНОГО ТУРНИРА ==========
function getTournamentParams() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Параметры');
  const data = sheet.getDataRange().getValues();
  
  // Параметры активного турнира — всегда в колонке B (индекс 1)
  const params = {};
  for (let i = 1; i < data.length; i++) {
    const key = data[i][0];
    const value = data[i][1];
    if (key && value) {
      params[key] = value;
    }
  }
  
  // Добавляем имя активного листа с матчами
  const activeSheetName = params['активный_лист'];
  if (activeSheetName) {
    params['активный_лист'] = activeSheetName;
  }
  
  return ContentService.createTextOutput(JSON.stringify(params))
    .setMimeType(ContentService.MimeType.JSON);
}

// ========== 2. ПОЛУЧЕНИЕ ДАННЫХ О СБОРНЫХ ==========
function getTeamsData() {
  // Определяем год активного турнира из листа Параметры
  const paramSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Параметры');
  const paramData = paramSheet.getDataRange().getValues();
  
  let currentYear = null;
  for (let i = 1; i < paramData.length; i++) {
    if (paramData[i][0] === 'турнир_год') {
      currentYear = parseInt(paramData[i][1]);
      break;
    }
  }
  
  if (!currentYear) {
    return ContentService.createTextOutput(JSON.stringify({}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  const teamsSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Сборные');
  const data = teamsSheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1);
  
  // Ищем колонку с рейтингом для нужного года
  const rankColumnName = `Рейтинг ${currentYear}`;
  let rankColumnIndex = -1;
  for (let i = 0; i < headers.length; i++) {
    if (headers[i] === rankColumnName) {
      rankColumnIndex = i;
      break;
    }
  }
  
  const teams = {};
  for (let i = 0; i < rows.length; i++) {
    const row = rows[i];
    const name = row[0];
    const flagCode = row[1];
    const rank = (rankColumnIndex !== -1) ? row[rankColumnIndex] : null;
    
    if (name) {
      teams[name] = {
        flagCode: flagCode,
        rank: rank ? parseInt(rank) : null
      };
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(teams))
    .setMimeType(ContentService.MimeType.JSON);
}

// ========== 3. ПОЛУЧЕНИЕ ДАННЫХ МАТЧЕЙ ==========
function getFullData() {
  // Получаем имя активного листа из параметров
  const paramSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Параметры');
  const paramData = paramSheet.getDataRange().getValues();
  
  let activeSheetName = null;
  for (let i = 1; i < paramData.length; i++) {
    if (paramData[i][0] === 'активный_лист') {
      activeSheetName = paramData[i][1];
      break;
    }
  }
  
  if (!activeSheetName) {
    return ContentService.createTextOutput(JSON.stringify({ error: 'Не указан активный_лист в параметрах' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(activeSheetName);
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({ error: 'Лист ' + activeSheetName + ' не найден' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  const allData = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
  
  const headers = allData[0];
  const rows = [];
  for (let i = 1; i < allData.length; i++) {
    const row = {};
    for (let j = 0; j < headers.length; j++) {
      row[headers[j]] = allData[i][j];
    }
    rows.push(row);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ headers: headers, rows: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ========== 4. СОХРАНЕНИЕ ПРОГНОЗОВ (POST) ==========
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const username = data.username;
    const predictions = data.predictions;
    const targetCol = data.targetCol;
    
    // Определяем активный лист
    const paramSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Параметры');
    const paramData = paramSheet.getDataRange().getValues();
    
    let activeSheetName = null;
    for (let i = 1; i < paramData.length; i++) {
      if (paramData[i][0] === 'активный_лист') {
        activeSheetName = paramData[i][1];
        break;
      }
    }
    
    if (!activeSheetName) {
      throw new Error('Не указан активный_лист в параметрах');
    }
    
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(activeSheetName);
    if (!sheet) throw new Error('Лист ' + activeSheetName + ' не найден');
    
    // Игнорируем targetCol, если он меньше 7 (некорректная колонка)
    let finalCol = (targetCol && targetCol >= 7) ? targetCol : null;
    
    if (!finalCol) {
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
      for (let i = 6; i < headers.length; i++) {
        if (headers[i] && headers[i].toString().startsWith('Участник')) {
          finalCol = i + 1;
          break;
        }
      }
    }
    
    if (!finalCol || finalCol < 7) {
      throw new Error('Не найдена колонка с "Участник"');
    }
    
    sheet.getRange(1, finalCol).setValue(username);
    
    const lastRow = sheet.getLastRow();
    const idRange = sheet.getRange(2, 1, lastRow - 1, 1);
    const idValues = idRange.getValues();
    
    const rowMap = {};
    for (let i = 0; i < idValues.length; i++) {
      const matchId = idValues[i][0];
      if (matchId) {
        rowMap[matchId] = i + 2;
      }
    }
    
    for (let i = 0; i < predictions.length; i++) {
      const matchId = predictions[i].match_id;
      const prediction = predictions[i].prediction;
      const rowIndex = rowMap[matchId];
      if (rowIndex) {
        sheet.getRange(rowIndex, finalCol).setValue(prediction);
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ success: true, targetCol: finalCol }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
